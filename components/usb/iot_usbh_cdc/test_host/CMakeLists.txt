cmake_minimum_required(VERSION 3.16)
project(iot_usbh_cdc_close_test LANGUAGES CXX)
enable_testing()

# Compile the real close/endpoint functions without the USB hardware and enumeration stack.
set(cdc_source "${CMAKE_CURRENT_SOURCE_DIR}/../iot_usbh_cdc.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${cdc_source}")
file(READ "${cdc_source}" cdc_code)
string(REPLACE "\r\n" "\n" cdc_code "${cdc_code}")
set(extracted "/* Generated from iot_usbh_cdc.c; do not edit. */\n")
foreach(function is_port_opened _cdc_stop_transfer_endpoint _cdc_reset_transfer_endpoint
                 _cdc_port_close usbh_cdc_port_close _cdc_close_disconnected_ports)
    string(REGEX MATCH "(static )?(void|bool|esp_err_t) ${function}\\([^\n]*\\)\n\\{" signature "${cdc_code}")
    if(NOT signature)
        message(FATAL_ERROR "CDC test cannot find definition of ${function}")
    endif()
    string(FIND "${cdc_code}" "${signature}" start)
    string(SUBSTRING "${cdc_code}" ${start} -1 body)
    string(FIND "${body}" "\n}\n" end)
    if(end LESS 0)
        message(FATAL_ERROR "CDC test cannot find end of ${function}")
    endif()
    math(EXPR length "${end} + 3")
    string(SUBSTRING "${body}" 0 ${length} body)
    string(APPEND extracted "${body}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cdc_close_under_test.inc" "${extracted}")

find_package(Threads REQUIRED)
add_executable(usb_cdc_close_test test_close.cpp)
target_compile_features(usb_cdc_close_test PRIVATE cxx_std_17)
target_include_directories(usb_cdc_close_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(usb_cdc_close_test PRIVATE Threads::Threads)
add_test(NAME usb_cdc_close_test COMMAND usb_cdc_close_test)
set_tests_properties(usb_cdc_close_test PROPERTIES TIMEOUT 15)

# Keep the checks active in Release builds too.
if(MSVC)
    target_compile_options(usb_cdc_close_test PRIVATE /UNDEBUG)
else()
    target_compile_options(usb_cdc_close_test PRIVATE -UNDEBUG)
endif()
